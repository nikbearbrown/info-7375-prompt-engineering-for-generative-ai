    25	# readabilipy >= 0.3.0 ships only `package.json` for its JavaScript and runs

    27	# (#4830). That stays the behaviour (the README states it), but the install now
    28	# runs at most once per process, under a lock and a timeout. It installs into

    30	# installed there ahead of time (the README says how) are found and nothing is
    31	# fetched.

   161	def _install_dependencies(jsdir: Path) -> bool:
   162	    """`npm install` readabilipy's JavaScript dependencies; whether it worked."""
   163	    npm = shutil.which("npm")
   164	    if npm is None:
   165	        _log(
   166	            "npm was not found on PATH, so Readability.js's dependencies cannot be "
   167	            f"installed in {jsdir}; using the pure-Python extractor"
   168	        )
   169	        return False
   170	    _log(
   171	        f"installing Readability.js's dependencies with npm in {jsdir} (first use; "
   172	        "this contacts the npm registry)"
   173	    )
   174	    done = _run(
   175	        [npm, "install", "--no-audit", "--no-fund"],
   176	        timeout=INSTALL_TIMEOUT_SECONDS,
   177	        cwd=jsdir,
   178	    )
   179	    if done is None:
   180	        return False
   181	    _forward(done.stdout)
   182	    _forward(done.stderr)
   183	    if done.returncode != 0 or not _dependencies_installed(jsdir):
   184	        _log(
   185	            f"npm install failed (exit {done.returncode}); using the pure-Python extractor"
   186	        )
   187	        return False
   188	    return True
